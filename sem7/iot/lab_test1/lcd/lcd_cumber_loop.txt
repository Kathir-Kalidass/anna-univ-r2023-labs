#include <8051.h>

unsigned char digits[] = {
    0x3F, 0x06, 0x5B, 0x4F, 0x66,
    0x6D, 0x7D, 0x07, 0x7F, 0x6F
};

void delay(void)
{
    unsigned int i;
    for (i = 0; i < 30000; i++);
}

void main(void)
{
    unsigned char i;

    while (1)
    {
        for (i = 0; i < 10; i++)
        {
            P2 = digits[i];
            delay();
        }
    }
}
